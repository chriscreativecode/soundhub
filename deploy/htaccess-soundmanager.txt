# Security Headers
<IfModule mod_headers.c>

    # Content Security Policy
    Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' https://www.googletagmanager.com https://www.googletagmanager.com/gtm.js; style-src 'self' 'unsafe-inline'; img-src 'self' data: https://www.googletagmanager.com https://www.google-analytics.com; font-src 'self'; frame-src https://www.googletagmanager.com; worker-src 'self'; connect-src 'self' https://www.googletagmanager.com https://www.google-analytics.com https://region1.google-analytics.com https://region1.analytics.google.com; form-action 'self'; base-uri 'self'; object-src 'none'; frame-ancestors 'none';"

    # HSTS
    Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"

    # Cross-Origin Opener Policy
    Header always set Cross-Origin-Opener-Policy "same-origin"

    # Clickjacking protection
    Header always set X-Frame-Options "DENY"

    # Prevents MIME-type sniffing
    Header always set X-Content-Type-Options "nosniff"

    # Controls referrer information
    Header always set Referrer-Policy "strict-origin-when-cross-origin"

    # Disable access to sensitive browser features
    Header always set Permissions-Policy "camera=(), microphone=(), geolocation=()"

</IfModule>

# Enable Gzip compression (fallback for older browsers)
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css text/javascript application/javascript application/json application/xml image/svg+xml font/woff font/woff2
</IfModule>

# Enable Brotli compression (preferred)
<IfModule mod_brotli.c>
    AddOutputFilterByType BROTLI text/html text/plain text/xml text/css text/javascript application/javascript application/json application/xml image/svg+xml font/woff font/woff2
</IfModule>

# Caching
#
# Rules are matched on the FILENAME, not the MIME type. Caching by type is what
# broke this before: "ExpiresByType application/javascript 1 year" also applied
# to the entry bundles, which at the time carried no content hash, so returning
# visitors kept running a year-old general.js after every deploy.
#
# The blocks below are ordered from broad to specific. Apache applies them in
# document order and a later "Header set" replaces an earlier one, so the
# fingerprint rule wins over the default, and HTML wins over everything.
<IfModule mod_headers.c>

    # 1. Default for files that keep their name across builds (favicons, logo,
    #    robots.txt, sitemap.xml, the manifest). Short, so an update is picked
    #    up the same day without hammering the server.
    <FilesMatch "\.(ico|png|jpe?g|svg|webp|gif|txt|xml|json|webmanifest|woff2?)$">
        Header set Cache-Control "public, max-age=3600"
    </FilesMatch>

    # 2. Fingerprinted build output: the filename carries a content hash, so any
    #    change produces a new URL. These can never go stale and are safe to keep
    #    forever.
    <FilesMatch "-[A-Za-z0-9_-]{8,}\.(js|css|mp3|wav|ogg|png|jpe?g|svg|webp|woff2?)$">
        Header set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>

    # 3. HTML is the entry point that names the hashed files above. If it goes
    #    stale the browser keeps requesting the previous build's assets, so it
    #    must be revalidated on every visit. A 304 makes that cheap.
    <FilesMatch "\.html$">
        Header set Cache-Control "no-cache, must-revalidate"
    </FilesMatch>

</IfModule>

# mod_expires would emit its own Expires header alongside the rules above and
# the two can disagree. Cache-Control takes precedence in every modern browser,
# so make sure no stale Expires header is left behind.
<IfModule mod_expires.c>
    ExpiresActive Off
</IfModule>
